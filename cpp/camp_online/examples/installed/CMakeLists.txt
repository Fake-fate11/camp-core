cmake_minimum_required(VERSION 3.16)
project(camp_installed_example LANGUAGES CXX)
option(CAMP_INSTALLED_MAP_BRIDGE "Also build a consumer of the installed map bridge" OFF)
if(CAMP_INSTALLED_MAP_BRIDGE)
  find_package(camp_online CONFIG REQUIRED COMPONENTS lanelet_context)
else()
  find_package(camp_online CONFIG REQUIRED)
endif()
add_executable(camp_installed_score ../score.cpp)
target_compile_features(camp_installed_score PRIVATE cxx_std_17)
target_link_libraries(camp_installed_score PRIVATE camp::camp_online nlohmann_json::nlohmann_json)

if(CAMP_INSTALLED_MAP_BRIDGE)
  add_executable(camp_installed_map map_consumer.cpp)
  target_link_libraries(camp_installed_map PRIVATE camp::camp_lanelet_context)
  enable_testing()
  add_test(NAME camp_installed_map COMMAND camp_installed_map)
endif()
