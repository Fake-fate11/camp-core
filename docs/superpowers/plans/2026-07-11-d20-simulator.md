# D20 Simulator Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a polished d20 simulator that runs offline by double-clicking one HTML file.

**Architecture:** A root-level standalone HTML document owns the markup, styling, random-number generation, animation, keyboard input, and in-memory history. A small standard-library Python test guards the single-file and interaction contracts without adding runtime dependencies.

**Tech Stack:** HTML5, CSS, browser JavaScript, Python `unittest`

## Global Constraints

- The simulator must run offline by double-clicking `d20-simulator.html`.
- Use no external assets, libraries, network requests, build tools, or storage.
- Generate final results with `crypto.getRandomValues()` and rejection sampling for an unbiased integer from 1 through 20.
- Support a real button, Space-key rolling, visible focus, and reduced-motion preferences.
- Prevent overlapping rolls and keep only the eight latest final results.

- [ ] **Step 1: Write the failing contract test**

Create `scripts/test_d20_simulator.py`:

```python
from pathlib import Path
import unittest


HTML_PATH = Path(__file__).resolve().parents[1] / "d20-simulator.html"


class D20SimulatorTest(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        cls.html = HTML_PATH.read_text(encoding="utf-8") if HTML_PATH.exists() else ""
        cls.lower = cls.html.lower()

    def test_is_one_offline_file(self):
        self.assertNotIn("<script src=", self.lower)
        self.assertNotIn("<link ", self.lower)
        self.assertNotIn("http://", self.lower)
        self.assertNotIn("https://", self.lower)

    def test_has_accessible_controls(self):
        self.assertTrue(HTML_PATH.exists(), "d20-simulator.html must exist")
        self.assertIn('type="button"', self.lower)
        self.assertIn('id="roll-button"', self.lower)
        self.assertIn('aria-live="polite"', self.lower)
        self.assertIn(':focus-visible', self.lower)

    def test_has_required_roll_contract(self):
        self.assertIn("crypto.getrandomvalues", self.lower)
        self.assertIn("4294967280", self.html)
        self.assertIn("event.code === \"Space\"", self.html)
        self.assertIn("slice(0, 8)", self.html)
        self.assertIn("prefers-reduced-motion", self.lower)
        self.assertIn("if (rolling) return", self.html)


if __name__ == "__main__":
    unittest.main()
```

    const die = document.querySelector("#die");
    const result = document.querySelector("#die-result");
    const status = document.querySelector("#status");
    const button = document.querySelector("#roll-button");
    const historyList = document.querySelector("#history-list");
    const reducedMotion = matchMedia("(prefers-reduced-motion: reduce)");
    const limit = 4294967280;
    const values = new Uint32Array(1);
    const rollHistory = [];
    let rolling = false;

    const sleep = milliseconds => new Promise(resolve => setTimeout(resolve, milliseconds));

    function randomD20() {
      do crypto.getRandomValues(values); while (values[0] >= limit);
      return values[0] % 20 + 1;
    }

    function renderHistory() {
      const items = rollHistory.slice(0, 8).map(value => {
        const item = document.createElement("li");
        item.textContent = value;
        return item;
      });
      historyList.replaceChildren(...items);
    }

    async function roll() {
      if (rolling) return;
      rolling = true;
      button.disabled = true;
      die.classList.add("rolling");
      die.setAttribute("aria-busy", "true");
      status.textContent = "命运转动中…";

      if (!reducedMotion.matches) {
        for (let index = 0; index < 8; index += 1) {
          result.textContent = randomD20();
          await sleep(55);
        }
      }

      const finalResult = randomD20();
      result.textContent = finalResult;
      rollHistory.unshift(finalResult);
      renderHistory();
      status.textContent = `本次结果：${finalResult}`;
      die.classList.remove("rolling");
      die.removeAttribute("aria-busy");
      button.disabled = false;
      rolling = false;
    }

    button.addEventListener("click", roll);
    document.addEventListener("keydown", event => {
      const target = event.target;
      const isInteractive = target instanceof HTMLElement &&
        (target.isContentEditable || /^(BUTTON|INPUT|SELECT|TEXTAREA|A)$/.test(target.tagName));
      if (event.code === "Space" && !isInteractive) {
        event.preventDefault();
        roll();
      }
    });

    console.assert(
      Array.from({ length: 100 }, randomD20).every(value => value >= 1 && value <= 20),
      "D20 results must stay between 1 and 20"
    );
  </script>
